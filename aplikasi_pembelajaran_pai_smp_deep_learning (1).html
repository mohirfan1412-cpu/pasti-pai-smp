    const AudioEngine = {

      ctx: null,

      init() {

        if (!this.ctx) {

          const AudioContext = window.AudioContext || window.webkitAudioContext;

          if (AudioContext) this.ctx = new AudioContext();

        }

      },

      playTone(freq, type = 'sine', duration = 0.15) {

        if (!userState.soundEnabled) return;

        try {

          this.init();

          if (!this.ctx) return;

          if (this.ctx.state === 'suspended') this.ctx.resume();

          const osc = this.ctx.createOscillator();

          const gain = this.ctx.createGain();

          osc.type = type;

          osc.frequency.setValueAtTime(freq, this.ctx.currentTime);

          gain.gain.setValueAtTime(0.12, this.ctx.currentTime);

          gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);

          osc.connect(gain);

          gain.connect(this.ctx.destination);

          osc.start();

          osc.stop(this.ctx.currentTime + duration);

        } catch (e) {}

      },

      playCollect() {

    function initPdfDocument() {

      const meta = BOOK_METADATA[bookReaderState.activeGrade];

      const spinner = document.getElementById('pdfLoadingSpinner');

      const canvas = document.getElementById('pdfCanvas');

      if (spinner) spinner.classList.add('hidden');

      if (canvas) canvas.classList.remove('hidden');


      if (bookReaderState.pdfDoc) {

        bookReaderState.pdfTotalPages = bookReaderState.pdfDoc.numPages;

        document.getElementById('pdfTotalPagesDisplay').textContent = bookReaderState.pdfTotalPages;

        document.getElementById('pdfPageInput').max = bookReaderState.pdfTotalPages;

        renderPdfPage(bookReaderState.pdfPageNum);

      } else {

        bookReaderState.pdfTotalPages = meta.totalPages;

        document.getElementById('pdfTotalPagesDisplay').textContent = meta.totalPages;

        document.getElementById('pdfPageInput').max = meta.totalPages;

        if (bookReaderState.pdfPageNum < 1 || bookReaderState.pdfPageNum > meta.totalPages) {

          bookReaderState.pdfPageNum = 1;

        }

        renderBuiltInBookPage(bookReaderState.pdfPageNum);

      }

    }


    function renderPdfPage(num) {

      if (!bookReaderState.pdfDoc) {

        renderBuiltInBookPage(num);

        return;

      }

      bookReaderState.pdfIsRendering = true;


      bookReaderState.pdfDoc.getPage(num).then(page => {

        const canvas = document.getElementById('pdfCanvas');

        if (!canvas) return;

        const ctx = canvas.getContext('2d');

        const viewport = page.getViewport({ scale: bookReaderState.pdfScale });


        canvas.height = viewport.height;

        canvas.width = viewport.width;


        const renderContext = {

          canvasContext: ctx,

          viewport: viewport

        };


        const renderTask = page.render(renderContext);

        renderTask.promise.then(() => {

          bookReaderState.pdfIsRendering = false;

          if (bookReaderState.pdfPageNumPending !== null) {

            renderPdfPage(bookReaderState.pdfPageNumPending);

            bookReaderState.pdfPageNumPending = null;

          }

        });

      }).catch(() => {

        bookReaderState.pdfIsRendering = false;

        renderBuiltInBookPage(num);

      });


      const pageInput = document.getElementById('pdfPageInput');

      if (pageInput) pageInput.value = num;

    }


    function queuePdfPage(num) {

      if (!bookReaderState.pdfDoc) {

        renderBuiltInBookPage(num);

        return;

      }

      if (bookReaderState.pdfIsRendering) {

        bookReaderState.pdfPageNumPending = num;

      } else {

        renderPdfPage(num);

      }

    }


    function changePdfPage(delta) {

      const total = bookReaderState.pdfTotalPages || BOOK_METADATA[bookReaderState.activeGrade].totalPages;

      const target = bookReaderState.pdfPageNum + delta;

      if (target >= 1 && target <= total) {

        bookReaderState.pdfPageNum = target;

        queuePdfPage(target);

    function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight) {

      const paragraphs = String(text || '').split('\n');

      let currentY = y;

      for (let p = 0; p < paragraphs.length; p++) {

        const words = paragraphs[p].split(' ');

        let line = '';

        for (let n = 0; n < words.length; n++) {

          const testLine = line + words[n] + ' ';

          const metrics = ctx.measureText(testLine);

          if (metrics.width > maxWidth && n > 0) {

            ctx.fillText(line.trim(), x, currentY);

            line = words[n] + ' ';

            currentY += lineHeight;

          } else {

            line = testLine;

          }

        }

        ctx.fillText(line.trim(), x, currentY);

        currentY += lineHeight * 1.25;

      }

      return currentY;

    }

    /**

     * Memanggil Gemini API (gemini-3-flash-preview) dengan mekanisme Exponential Backoff

     */

    async function callGeminiAPI(payload, maxRetries = 4) {

      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

      let delay = 1000;


      for (let attempt = 0; attempt <= maxRetries; attempt++) {

        try {

          const response = await fetch(apiUrl, {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify(payload)

          });


          if (!response.ok) {

            throw new Error(`HTTP status ${response.status}`);

          }


          const result = await response.json();

          const text = result?.candidates?.[0]?.content?.parts?.[0]?.text;

          if (!text) {

            throw new Error("Respons AI kosong");

          }

          return text;

        } catch (err) {

          if (attempt === maxRetries) {

            throw err;

          }

          await new Promise(resolve => setTimeout(resolve, delay));

          delay *= 2;

        }

      }

    }

    function formatAITextToHTML(rawText) {

      if (!rawText) return "";

      let safe = rawText

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;");


      // Format bold **teks**

      safe = safe.replace(/\*\*(.*?)\*\*/g, '<strong class="font-bold text-slate-900">$1</strong>');

      // Format italic *teks*

      safe = safe.replace(/\*(.*?)\*/g, '<em>$1</em>');

      // Format bullet list

      safe = safe.replace(/^\s*[-•]\s+(.*)$/gm, '<li class="ml-4 list-disc">$1</li>');

      // Format numbered list

      safe = safe.replace(/^\s*(\d+)\.\s+(.*)$/gm, '<div class="flex items-start space-x-1.5 mt-1"><span class="font-bold text-emerald-700">$1.</span><span>$2</span></div>');

      // Convert double line breaks to paragraphs

      safe = safe.replace(/\n\n/g, '<br><br>').replace(/\n/g, '<br>');

      return safe;

    }

    function updateGame() {

      const p = gameState.player;

      if (gameState.keys.left) p.vx = -p.speed;

      else if (gameState.keys.right) p.vx = p.speed;

      else p.vx *= 0.75;


      if (gameState.keys.up && p.onGround) {

        p.vy = gameState.jumpPower;

        p.onGround = false;

        AudioEngine.playJump();

      }


      p.vy += gameState.gravity;

      p.x += p.vx;

      p.y += p.vy;


      p.onGround = false;

      gameState.platforms.forEach(plat => {

        if (p.x + p.w > plat.x && p.x < plat.x + plat.w &&

            p.y + p.h >= plat.y && p.y + p.h <= plat.y + 14 && p.vy >= 0) {

          p.y = plat.y - p.h;

          p.vy = 0;

          p.onGround = true;

        }

      });


      if (p.x < 0) p.x = 0;

      if (p.x + p.w > 900) p.x = 900 - p.w;


      gameState.gems.forEach(g => {

        if (!g.collected && Math.hypot((p.x + p.w/2) - g.x, (p.y + p.h/2) - g.y) < 24) {

          g.collected = true;

          gameState.score += 20;

          addXP(20);

          document.getElementById('gameScore').textContent = gameState.score;

          AudioEngine.playCollect();

          showToast(`Butir Kebajikan: ${g.name} (+20 XP)!`, 'success');

        }

      });


      gameState.hazards.forEach(h => {

        if (p.x + p.w > h.x && p.x < h.x + h.w && p.y + p.h > h.y && p.y < h.y + h.h) {

          gameState.hearts--;

    function handleQuizAnswer(isCorrect) {

    function saveJournalEntry() {

      const cat = document.getElementById('journalCategory').value;

      const input = document.getElementById('journalInput');

      if (!input || !input.value.trim()) {

        showToast("Tuliskan refleksi terlebih dahulu!", "error");

        return;

      }


      const newEntry = {

        id: Date.now(),

        category: cat,

        date: "Hari Ini",

        text: input.value.trim()

      };


      userState.journals.unshift(newEntry);

      input.value = "";

      renderJournalList();

      addXP(30);

      AudioEngine.playSuccess();

      showToast("Alhamdulillah! Catatan muhasabah tersimpan (+30 XP).", "success");

    }


    function deleteJournalEntry(id) {

      userState.journals = userState.journals.filter(j => j.id !== id);

      renderJournalList();

      showToast("Catatan dihapus", "info");

    }
